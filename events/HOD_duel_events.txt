add_namespace = hod_duel

# HOD duels — reports for automatic battlefield duels (common/scripted_effects/HOD_duel_scripted_effects.txt).
# Event targets: hod_duel_winner / hod_duel_loser (or duel_char1_token / duel_char2_token on a stalemate).
# The texts follow the fighter's style flag, stamped on its owner by HOD_stamp_duel_*_style.

########################################
# hod_duel.9 — hidden, winner's owner: capture a decisively beaten loser, then send both reports
# (capture_by needs the captor country as ROOT)
########################################
country_event = {
	id = hod_duel.9
	hidden = yes
	is_triggered_only = yes

	immediate = {
		event_target:hod_duel_loser = {
			if = {
				limit = { has_unit_leader_flag = hod_duel_decisive }
				if = {
					limit = { can_be_captured = yes }
					capture_by = ROOT
				}
				else = {
					add_timed_unit_leader_trait = { trait = wounded days = 120 }
				}
			}
		}
		event_target:hod_duel_winner = { unit_leader_event = { id = hod_duel.1 } }
		event_target:hod_duel_loser = { unit_leader_event = { id = hod_duel.2 days = 1 } }
	}
}
