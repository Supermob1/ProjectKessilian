add_namespace = hod_duel

#####################################################################
# Duel system — flag-only architecture (no event targets).
#
# Why no event targets:
#   * save_global_event_target_as persists across event firings, BUT
#     it's a single global slot that two simultaneous duels would
#     overwrite. With days = 1 delays between hod_duel.1 → .2 → .3 → .4,
#     ample time exists for another country to start its own duel and
#     trash both targets. The "Trying to remove undefined event target"
#     errors come from hod_duel.4 firing more than once for one duel
#     (e.g. stale enemy_fighter flag) or from cross-duel interference.
#
# Architecture:
#   * has_global_flag = hod_duel_active           — only ONE duel runs globally
#   * country flag    = hod_duel_initiator_country — set on the initiator
#   * country flag    = hod_duel_is_enemy_country  — set on the enemy
#   * character flag  = hod_duel_own_fighter       — the initiator's chosen fighter
#   * character flag  = hod_duel_enemy_fighter     — the enemy's chosen fighter
#   * character flags = hod_duel_own_offensive/defensive,
#                       hod_duel_enemy_offensive/defensive — chosen stances
#   * country flags   = hod_duel_tactic_*          — drives hod_duel.2/3 flavor desc
#   * country flags   = hod_duel_style_*           — drives outcome event flavor desc
#
# All cross-scope lookups use:
#   every_country = { limit = { has_country_flag = ... } ... }     in effects
#   any_country   = { has_country_flag = ... any_character = ... } in triggers
#####################################################################

########################################
# hod_duel.1 — Duel opportunity (auto, country_event)
########################################

country_event = {
	id = hod_duel.1
	title = hod_duel.1.t
	picture = GFX_report_event_jedi_duel

	desc = {
		trigger = {
			any_character = { has_character_flag = hod_duel_own_fighter has_trait = force_sensitive }
			any_country = {
				has_country_flag = hod_duel_is_enemy_country
				any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = force_sensitive }
			}
		}
		text = hod_duel.1.d_force_vs_force
	}
	desc = {
		trigger = {
			any_character = { has_character_flag = hod_duel_own_fighter has_trait = force_sensitive }
			any_country = {
				has_country_flag = hod_duel_is_enemy_country
				any_character = { has_character_flag = hod_duel_enemy_fighter NOT = { has_trait = force_sensitive } }
			}
		}
		text = hod_duel.1.d_force_vs_warrior
	}
	desc = {
		trigger = {
			any_character = { has_character_flag = hod_duel_own_fighter NOT = { has_trait = force_sensitive } }
			any_country = {
				has_country_flag = hod_duel_is_enemy_country
				any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = force_sensitive }
			}
		}
		text = hod_duel.1.d_warrior_vs_force
	}
	desc = hod_duel.1.d  # fallback: warrior vs warrior

	trigger = {
		has_war = yes
		NOT = { has_global_flag = hod_duel_active }
		any_character = { is_assigned = yes }
		any_enemy_country = { any_character = { is_assigned = yes } }
	}

	mean_time_to_happen = { days = 60 }

	immediate = {
		# Serialize duels — only one active globally.
		set_global_flag = hod_duel_active

		# Defensive cleanup — wipe ALL prior duel state from every country and character.
		# Worst case this clears one duel's stale half-state; best case it does nothing.
		every_country = {
			clr_country_flag = hod_duel_initiator_country
			clr_country_flag = hod_duel_is_enemy_country
			clr_country_flag = hod_duel_in_progress
			clr_country_flag = hod_duel_tactic_mando
			clr_country_flag = hod_duel_tactic_gunfighter
			clr_country_flag = hod_duel_tactic_dirty
			clr_country_flag = hod_duel_tactic_warrior
			clr_country_flag = hod_duel_tactic_soresu
			clr_country_flag = hod_duel_tactic_aggressive
			clr_country_flag = hod_duel_tactic_force
			clr_country_flag = hod_duel_tactic_cyborg
			clr_country_flag = hod_duel_tactic_dark
			every_character = {
				clr_character_flag = hod_duel_own_fighter
				clr_character_flag = hod_duel_own_offensive
				clr_character_flag = hod_duel_own_defensive
				clr_character_flag = hod_duel_enemy_fighter
				clr_character_flag = hod_duel_enemy_offensive
				clr_character_flag = hod_duel_enemy_defensive
			}
		}

		# Mark initiator (current scope = the country firing hod_duel.1).
		set_country_flag = hod_duel_in_progress
		set_country_flag = hod_duel_initiator_country

		# Pick own fighter — Force-sensitives > skilled_duelists > any assigned commander
		if = {
			limit = { any_character = { is_assigned = yes has_trait = force_sensitive } }
			random_character = {
				limit = { is_assigned = yes has_trait = force_sensitive }
				set_character_flag = hod_duel_own_fighter
			}
		}
		else_if = {
			limit = { any_character = { is_assigned = yes has_trait = skilled_duelist } }
			random_character = {
				limit = { is_assigned = yes has_trait = skilled_duelist }
				set_character_flag = hod_duel_own_fighter
			}
		}
		else = {
			random_character = {
				limit = { is_assigned = yes }
				set_character_flag = hod_duel_own_fighter
			}
		}

		# Pick enemy country and enemy fighter.
		random_enemy_country = {
			limit = { any_character = { is_assigned = yes } }
			set_country_flag = hod_duel_is_enemy_country
			save_global_event_target_as = hod_duel_enemy_country_ref
			if = {
				limit = { any_character = { is_assigned = yes has_trait = force_sensitive } }
				random_character = {
					limit = { is_assigned = yes has_trait = force_sensitive }
					set_character_flag = hod_duel_enemy_fighter
				}
			}
			else_if = {
				limit = { any_character = { is_assigned = yes has_trait = skilled_duelist } }
				random_character = {
					limit = { is_assigned = yes has_trait = skilled_duelist }
					set_character_flag = hod_duel_enemy_fighter
				}
			}
			else = {
				random_character = {
					limit = { is_assigned = yes }
					set_character_flag = hod_duel_enemy_fighter
				}
			}
		}

		# Save the fighter characters as global event targets so loc can use
		# [hod_duel_own_char.GetName] and [hod_duel_enemy_char.GetName].
		# The hod_duel_active global flag prevents overlapping duels, so these are safe.
		# Initiator country also saved as event target for loc references.
		save_global_event_target_as = hod_duel_initiator_country_ref
		every_character = {
			limit = { has_character_flag = hod_duel_own_fighter }
			save_global_event_target_as = hod_duel_own_char
		}
		every_country = {
			limit = { has_country_flag = hod_duel_is_enemy_country }
			every_character = {
				limit = { has_character_flag = hod_duel_enemy_fighter }
				save_global_event_target_as = hod_duel_enemy_char
			}
		}
	}

	# Engage — fire the own fighter's tactic-choice event
	option = {
		name = hod_duel.1.a
		every_character = {
			limit = { has_character_flag = hod_duel_own_fighter }
			unit_leader_event = { id = hod_duel.2 days = 1 }
		}
		ai_chance = { factor = 80 }
	}

	# Stand down — full state wipe
	option = {
		name = hod_duel.1.b
		clr_global_flag = hod_duel_active
		clear_global_event_target = hod_duel_own_char
		clear_global_event_target = hod_duel_enemy_char
		clear_global_event_target = hod_duel_initiator_country_ref
		clear_global_event_target = hod_duel_enemy_country_ref
		every_country = {
			limit = {
				OR = {
					has_country_flag = hod_duel_initiator_country
					has_country_flag = hod_duel_is_enemy_country
					has_country_flag = hod_duel_in_progress
				}
			}
			clr_country_flag = hod_duel_initiator_country
			clr_country_flag = hod_duel_is_enemy_country
			clr_country_flag = hod_duel_in_progress
			every_character = {
				limit = {
					OR = {
						has_character_flag = hod_duel_own_fighter
						has_character_flag = hod_duel_enemy_fighter
					}
				}
				clr_character_flag = hod_duel_own_fighter
				clr_character_flag = hod_duel_enemy_fighter
			}
		}
		ai_chance = { factor = 20 }
	}
}

########################################
# hod_duel.2 — Own fighter's tactic choice
# unit_leader_event for the own fighter
########################################

unit_leader_event = {
	id = hod_duel.2
	title = hod_duel.2.t
	picture = GFX_report_event_jedi_duel
	is_triggered_only = yes

	immediate = {
		every_character = {
			limit = { has_character_flag = hod_duel_own_fighter }
			if = {
				limit = { NOT = { has_trait = force_sensitive } has_trait = mandalorian_combat }
				owner = { set_country_flag = hod_duel_tactic_mando }
			}
			else_if = {
				limit = { NOT = { has_trait = force_sensitive } has_trait = blaster_duelist }
				owner = { set_country_flag = hod_duel_tactic_gunfighter }
			}
			else_if = {
				limit = { NOT = { has_trait = force_sensitive } has_trait = dirty_fighter }
				owner = { set_country_flag = hod_duel_tactic_dirty }
			}
			else_if = {
				limit = { NOT = { has_trait = force_sensitive } }
				owner = { set_country_flag = hod_duel_tactic_warrior }
			}
			else_if = {
				limit = {
					has_trait = force_sensitive
					OR = { has_trait = form_soresu has_trait = form_soresu_mastery has_trait = force_barrier has_trait = force_barrier_mastery }
				}
				owner = { set_country_flag = hod_duel_tactic_soresu }
			}
			else_if = {
				limit = {
					has_trait = force_sensitive
					OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_djem_so has_trait = form_djem_so_mastery }
				}
				owner = { set_country_flag = hod_duel_tactic_aggressive }
			}
			else = { owner = { set_country_flag = hod_duel_tactic_force } }
		}
	}

	desc = { trigger = { has_country_flag = hod_duel_tactic_mando      } text = hod_duel.2.d_mandalorian  }
	desc = { trigger = { has_country_flag = hod_duel_tactic_gunfighter  } text = hod_duel.2.d_gunfighter   }
	desc = { trigger = { has_country_flag = hod_duel_tactic_dirty       } text = hod_duel.2.d_dirty_fighter }
	desc = { trigger = { has_country_flag = hod_duel_tactic_warrior     } text = hod_duel.2.d_warrior       }
	desc = { trigger = { has_country_flag = hod_duel_tactic_soresu      } text = hod_duel.2.d_soresu        }
	desc = { trigger = { has_country_flag = hod_duel_tactic_aggressive  } text = hod_duel.2.d_aggressive    }
	desc = hod_duel.2.d  # Force-generic + safety fallback

	# Strike first — offensive
	option = {
		name = hod_duel.2.a
		HOD_clr_duel_tactic_flags = yes
		every_character = {
			limit = { has_character_flag = hod_duel_own_fighter }
			set_character_flag = hod_duel_own_offensive
		}
		every_country = {
			limit = { has_country_flag = hod_duel_is_enemy_country }
			every_character = {
				limit = { has_character_flag = hod_duel_enemy_fighter }
				unit_leader_event = { id = hod_duel.3 days = 1 }
			}
		}
		ai_chance = { factor = 40 }
	}

	# Read their movements — defensive
	option = {
		name = hod_duel.2.b
		HOD_clr_duel_tactic_flags = yes
		every_character = {
			limit = { has_character_flag = hod_duel_own_fighter }
			set_character_flag = hod_duel_own_defensive
		}
		every_country = {
			limit = { has_country_flag = hod_duel_is_enemy_country }
			every_character = {
				limit = { has_character_flag = hod_duel_enemy_fighter }
				unit_leader_event = { id = hod_duel.3 days = 1 }
			}
		}
		ai_chance = { factor = 30 }
	}

	# Trust your instincts — balanced
	option = {
		name = hod_duel.2.c
		HOD_clr_duel_tactic_flags = yes
		every_country = {
			limit = { has_country_flag = hod_duel_is_enemy_country }
			every_character = {
				limit = { has_character_flag = hod_duel_enemy_fighter }
				unit_leader_event = { id = hod_duel.3 days = 1 }
			}
		}
		ai_chance = { factor = 30 }
	}
}

########################################
# hod_duel.4 — Resolution (hidden, country_event for initiator)
########################################

country_event = {
	id = hod_duel.4
	hidden = yes
	is_triggered_only = yes

	# Guard against double-firing: must still have initiator flag set.
	# If a previous firing already cleaned up state, this is a no-op.
	trigger = { has_country_flag = hod_duel_initiator_country }

	immediate = {
		# Re-clear initiator flag immediately so a second hod_duel.4 firing
		# (from a stale enemy fighter) will fail the trigger and do nothing.
		clr_country_flag = hod_duel_initiator_country
	}

	option = {
		name = hod_duel.4.a

		random_list = {

			# ── Own fighter wins ──────────────────────────────────
			40 = {
				modifier = {
					factor = 1.5
					any_character = {
						has_character_flag = hod_duel_own_fighter
						OR = { has_trait = form_ataru_mastery has_trait = form_djem_so_mastery has_trait = force_push_mastery }
					}
				}
				modifier = {
					factor = 1.3
					any_character = { has_character_flag = hod_duel_own_fighter has_character_flag = hod_duel_own_offensive }
				}
				modifier = {
					factor = 0.6
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = jedi_padawan_rank }
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = dark_side_user }
					}
				}
				modifier = {
					factor = 1.2
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = jedi_veteran_rank }
				}
				modifier = {
					factor = 1.2
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = force_prescience_mastery }
				}
				modifier = {
					factor = 0.8
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_character_flag = hod_duel_enemy_offensive }
					}
				}
				modifier = {
					factor = 1.4
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = force_sensitive }
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter NOT = { has_trait = force_sensitive } }
					}
				}
				modifier = {
					factor = 0.7
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = combat_cyborg }
					}
				}
				modifier = {
					factor = 0.5
					any_character = { has_character_flag = hod_duel_own_fighter NOT = { has_trait = force_sensitive } }
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = force_sensitive }
					}
				}
				modifier = {
					factor = 1.2
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = mandalorian_combat }
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = force_sensitive }
					}
				}
				modifier = {
					factor = 1.3
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = blaster_duelist_mastery }
				}
				modifier = {
					factor = 1.2
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = dirty_fighter_mastery }
				}
				modifier = {
					factor = 1.15
					any_country = {
						has_country_flag = hod_duel_is_enemy_country
						any_character = { has_character_flag = hod_duel_enemy_fighter has_trait = droid_tactical_mastery }
					}
					any_character = { has_character_flag = hod_duel_own_fighter has_trait = force_sensitive }
				}

				every_character = {
					limit = { has_character_flag = hod_duel_own_fighter }
					HOD_stamp_duel_victory_style = yes
					HOD_add_duel_xp_winner = yes
					unit_leader_event = { id = hod_duel.10 days = 1 }
					clr_character_flag = hod_duel_own_offensive
					clr_character_flag = hod_duel_own_defensive
					clr_character_flag = hod_duel_own_fighter
				}
				every_country = {
					limit = { has_country_flag = hod_duel_is_enemy_country }
					every_character = {
						limit = { has_character_flag = hod_duel_enemy_fighter }
						HOD_stamp_duel_defeat_style = yes
						HOD_add_duel_xp_loser = yes
						unit_leader_event = { id = hod_duel.21 days = 1 }
						clr_character_flag = hod_duel_enemy_offensive
						clr_character_flag = hod_duel_enemy_defensive
						clr_character_flag = hod_duel_enemy_fighter
					}
					clr_country_flag = hod_duel_is_enemy_country
				}
				clr_country_flag = hod_duel_in_progress
				clr_global_flag = hod_duel_active
			}

			# ── Enemy fighter wins ──────────────────────────────────
			40 = {
				modifier = {
					factor = 1.6
					any_country = {
