# While duel_outcome == 0: options A/B/C visible, player picks move.
# When duel_outcome > 0: only option D ("Acknowledge") shows;
#                       paragraphs 4 & 5 (finale + consequences) fill in.
########################################
country_event = {
	id = hod_duel.2
	title = hod_duel.2.t
	picture = GFX_report_event_jedi_duel

	desc = hod_duel.2.d

	is_triggered_only = yes

	# NO `after` block. The old `after = { if duel_outcome = 0: fire .201 }`
	# misfired after option D's cleanup zeroed duel_outcome, causing the
	# .2 → .201 → .3 loop. Next-event firing is now inline in options A/B/C.

	# ── Round move options (only while duel ongoing) ─────────
	option = {
		name = hod_duel.2.a  # Attack
		trigger = { check_variable = { duel_outcome = 0 } }
		event_target:duel_char1_token = { HOD_duel_set_move_attack = yes }
		event_target:duel_char2_token.owner = { country_event = { id = hod_duel.201 } }
		custom_effect_tooltip = show_current_variables
		ai_chance = {
			factor = 50
			modifier = {
				factor = 1.5
				event_target:duel_char1_token = {
					OR = {
						has_trait = form_juyo_vaapad
						has_trait = form_ataru
						has_trait = form_ataru_mastery
						has_trait = form_djem_so_mastery
						has_trait = mandalorian_combat
						has_trait = dirty_fighter
					}
				}
			}
		}
	}
	option = {
		name = hod_duel.2.b  # Defend
		trigger = { check_variable = { duel_outcome = 0 } }
		event_target:duel_char1_token = { HOD_duel_set_move_defend = yes }
		event_target:duel_char2_token.owner = { country_event = { id = hod_duel.201 } }

			factor = 30

				factor = 1.5
				event_target:duel_char1_token = {
					OR = {
						has_trait = form_soresu
						has_trait = form_soresu_mastery
						has_trait = form_niman

						has_trait = heavy_armor_plating

########################################
# hod_duel.4 — Resolution (hidden)
#   Reads edge, picks outcome. Both fighters get the appropriate
#   outcome event with style flavor based on traits, plus a wound
#   modifier if their risk built too high.
########################################
country_event = {
	id = hod_duel.4
	hidden = yes
	is_triggered_only = yes

	immediate = {
		# Own wins
		if = {
			limit = { check_variable = { hod_duel_edge > 1 } }
			# Wound the winner if they overextended
			if = {
				limit = { check_variable = { hod_duel_risk_own > 4 } }
				event_target:hod_duel_own_country = { set_country_flag = hod_duel_winner_wounded }
			}
			# Stamp styles & XP, fire outcome events
			event_target:hod_duel_own_char = {
				HOD_stamp_duel_victory_style = yes
				HOD_add_duel_xp_winner = yes
				unit_leader_event = { id = hod_duel.10 days = 1 }
			}
			event_target:hod_duel_enemy_char = {
				HOD_stamp_duel_defeat_style = yes
				HOD_add_duel_xp_loser = yes
				unit_leader_event = { id = hod_duel.21 days = 1 }
			}
		}
		# Enemy wins
		else_if = {
			limit = { check_variable = { hod_duel_edge < -1 } }
			if = {
				limit = { check_variable = { hod_duel_risk_enemy > 4 } }
				event_target:hod_duel_enemy_country = { set_country_flag = hod_duel_winner_wounded }
			}
			event_target:hod_duel_own_char = {
				HOD_stamp_duel_defeat_style = yes
				HOD_add_duel_xp_loser = yes
				unit_leader_event = { id = hod_duel.11 days = 1 }
			}
			event_target:hod_duel_enemy_char = {
				HOD_stamp_duel_victory_style = yes
				HOD_add_duel_xp_winner = yes
				unit_leader_event = { id = hod_duel.20 days = 1 }
			}
		}
		# Stalemate
		else = {
			event_target:hod_duel_own_char = {
				HOD_stamp_duel_stale_style = yes
				HOD_add_duel_xp_stalemate = yes
				unit_leader_event = { id = hod_duel.12 days = 1 }
			}
			event_target:hod_duel_enemy_char = {
				HOD_stamp_duel_stale_style = yes
				HOD_add_duel_xp_stalemate = yes
				unit_leader_event = { id = hod_duel.22 days = 1 }
			}
		}

		HOD_duel_cleanup = yes
	}

	option = { name = hod_duel.4.a }
}
