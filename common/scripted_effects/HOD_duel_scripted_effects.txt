#####################################################################
# HOD duels — automatic battlefield duels
# ───────────────────────────────────────────────────────────────────
# On every battle a duelist's army fights (common/on_actions/HOD_duel_on_actions.txt): if an enemy duelist's Army HQ
# stands in the same or a neighbouring state, the battle turns into a duel 25% of the time. HOD_is_duelist: a Force user
# or champion fighter. The side that won the battle fights with momentum (+5 attack).
# Three rounds are fought automatically with HOD_duel_calculation_base (attack / defence / luck from ranks,
# lightsaber forms, Force powers and champion traits, plus form clashes and luck rolls).
#   won 3-0 or 2-0  decisive: the loser is captured (capture_by; held in one of the winner's provinces),
#                   or wounded 120 days if they can't be captured. Nobody dies in a duel.
#   won 2-1         the loser is wounded 60 days
#   even            both are wounded 20 days
# Winner: duel XP (ranks, forms, masteries) and the hod_duel_victor trait for 90 days. Loser: some XP.
# Both fighters rest 60 days (hod_duel_cooldown). Each owner gets a report event (hod_duel.1/.2/.3) whose text
# follows the fighter's style (lightsaber form, dark side, Mandalorian, droid...).
#####################################################################

# Unit leader scope, from on_army_leader_won_combat / _lost_combat (temp variable hod_duel_momentum = 1 / -1)
HOD_duel_on_battle = {
	if = {
		limit = { HOD_is_duelist = yes }
		# flag the state this leader's HQ stands in and its neighbours (nothing if the leader isn't deployed):
		# the enemy HQ is often across the border in an attack
		random_state = {
			limit = { PREV = { is_in_state = PREV } }
			set_state_flag = hod_duel_here
			every_neighbor_state = { set_state_flag = hod_duel_here }
		}
		set_temp_variable = { hod_duel_found = 0 }
		save_event_target_as = duel_char1_token
		owner = {
			random_enemy_country = {
				limit = {
					any_army_leader = {
						HOD_is_duelist = yes
						HOD_duel_hq_in_flagged_state = yes
					}
				}
				random_army_leader = {
					limit = {
						HOD_is_duelist = yes
						HOD_duel_hq_in_flagged_state = yes
					}
					save_event_target_as = duel_char2_token
					set_temp_variable = { hod_duel_found = 1 }
				}
			}
		}
		every_state = {
			limit = { has_state_flag = hod_duel_here }
			clr_state_flag = hod_duel_here
		}
		if = {
			limit = { check_variable = { hod_duel_found = 1 } }
			random_list = {
				25 = { owner = { HOD_duel_fight = yes } }
				75 = { }
			}
		}
	}
}

# Country scope (owner of duel_char1_token); event targets duel_char1_token / duel_char2_token are set
HOD_duel_fight = {
	event_target:duel_char1_token = {
		HOD_duel_setup_calculate_powers = yes
		set_variable = { duel_rounds_won = 0 }
		set_variable = { hod_duel_round = 0 }        # HOD_duel_calculation_base reads it (late-round effects)
		set_unit_leader_flag = { flag = hod_duel_cooldown value = 1 days = 60 }
	}
	event_target:duel_char2_token = {
		HOD_duel_setup_calculate_powers = yes
		set_variable = { duel_rounds_won = 0 }
		set_variable = { hod_duel_round = 0 }        # HOD_duel_calculation_base reads it (late-round effects)
		set_unit_leader_flag = { flag = hod_duel_cooldown value = 1 days = 60 }
	}
	# momentum: the side that just won the battle (hod_duel_momentum 1 = char1's side, -1 = char2's; 0 from debug)
	if = {
		limit = { check_variable = { hod_duel_momentum = 1 } }
		event_target:duel_char1_token = { add_to_variable = { att = 5 } }
	}
	else_if = {
		limit = { check_variable = { hod_duel_momentum = -1 } }
		event_target:duel_char2_token = { add_to_variable = { att = 5 } }
	}
	HOD_duel_auto_round = yes
	HOD_duel_auto_round = yes
	HOD_duel_auto_round = yes

	set_temp_variable = { hod_duel_margin = event_target:duel_char1_token.duel_rounds_won }
	subtract_from_temp_variable = { hod_duel_margin = event_target:duel_char2_token.duel_rounds_won }
	if = {
		limit = { check_variable = { hod_duel_margin > 0 } }
		event_target:duel_char1_token = { save_event_target_as = hod_duel_winner }
		event_target:duel_char2_token = { save_event_target_as = hod_duel_loser }
	}
	else_if = {
		limit = { check_variable = { hod_duel_margin < 0 } }
		event_target:duel_char2_token = { save_event_target_as = hod_duel_winner }
		event_target:duel_char1_token = { save_event_target_as = hod_duel_loser }
		multiply_temp_variable = { hod_duel_margin = -1 }
	}

	if = {
		limit = { check_variable = { hod_duel_margin = 0 } }
		event_target:duel_char1_token.owner = { HOD_clr_duel_style_flags = yes }
		event_target:duel_char2_token.owner = { HOD_clr_duel_style_flags = yes }
		event_target:duel_char1_token = {
			HOD_stamp_duel_stale_style = yes
			HOD_add_duel_xp_stalemate = yes
			add_timed_unit_leader_trait = { trait = wounded days = 20 }
		}
		event_target:duel_char2_token = {
			HOD_stamp_duel_stale_style = yes
			HOD_add_duel_xp_stalemate = yes
			add_timed_unit_leader_trait = { trait = wounded days = 20 }
		}
		event_target:duel_char1_token = { unit_leader_event = { id = hod_duel.3 } }
		event_target:duel_char2_token = { unit_leader_event = { id = hod_duel.3 } }
	}
	else = {
		event_target:duel_char1_token.owner = { HOD_clr_duel_style_flags = yes }
		event_target:duel_char2_token.owner = { HOD_clr_duel_style_flags = yes }
		event_target:hod_duel_winner = {
			HOD_stamp_duel_victory_style = yes
			HOD_add_duel_xp_winner = yes
			add_timed_unit_leader_trait = { trait = hod_duel_victor days = 90 }
		}
		event_target:hod_duel_loser = {
			HOD_stamp_duel_defeat_style = yes
			HOD_add_duel_xp_loser = yes
			if = {
				limit = { check_variable = { hod_duel_margin > 1 } }
				set_unit_leader_flag = { flag = hod_duel_decisive value = 1 days = 7 }   # hod_duel.9 captures
			}
			else = {
				add_timed_unit_leader_trait = { trait = wounded days = 60 }
			}
		}
		# hod_duel.9 (hidden, winner's owner): captures a decisively beaten loser, then sends both reports
		event_target:hod_duel_winner.owner = { country_event = { id = hod_duel.9 } }
	}
}

# One automatic round: HOD_duel_calculation_base sets char1_power / char2_power in this scope
HOD_duel_auto_round = {
	HOD_duel_calculation_base = yes
	if = {
		limit = { check_variable = { char1_power > char2_power } }
		event_target:duel_char1_token = { add_to_variable = { duel_rounds_won = 1 } }
	}
	else_if = {
		limit = { check_variable = { char2_power > char1_power } }
		event_target:duel_char2_token = { add_to_variable = { duel_rounds_won = 1 } }
	}
	event_target:duel_char1_token = { add_to_variable = { hod_duel_round = 1 } }
	event_target:duel_char2_token = { add_to_variable = { hod_duel_round = 1 } }
}

#####################################################################
# HOD_duel_calculation_base
# ───────────────────────────────────────────────────────────────────
# Per-round resolution. Computes char1_power and char2_power; the
# higher value is winning that round. Layered phases:
#   Phase 1 — Base contest: my att vs your def
#   Phase 2 — Luck roll: 20/60/20 bad/avg/crit, scales with luc
#   Phase 3 — Form clash: rock-paper-scissors lightsaber styles
#   Phase 4 — Force power interactions
#   Phase 5 — Non-Force & equipment interactions
#   Phase 6 — Rank pressure
# Call from event scope (initiator country). Reads variables from
# both fighter event targets.
#####################################################################
HOD_duel_calculation_base = {
    set_variable = { char1_power = 0 }
    set_variable = { char2_power = 0 }

    # ═════════════════════════════════════════════════════════════
    # PHASE 1 — BASE CONTEST
    # My attack potential, soaked by half of opponent's defense.
    # ═════════════════════════════════════════════════════════════
    add_to_variable = { char1_power = event_target:duel_char1_token.att }
    set_temp_variable = { temp_soak = event_target:duel_char2_token.def }
    multiply_temp_variable = { temp_soak = 0.5 }
    subtract_from_variable = { char1_power = temp_soak }

    add_to_variable = { char2_power = event_target:duel_char2_token.att }
    set_temp_variable = { temp_soak = event_target:duel_char1_token.def }
    multiply_temp_variable = { temp_soak = 0.5 }
    subtract_from_variable = { char2_power = temp_soak }

    # ═════════════════════════════════════════════════════════════
    # PHASE 2 — LUCK ROLL (per character)
    # 20% unlucky / 60% average / 20% critical hit.
    # Higher luc = bigger swing in both directions.
    # ═════════════════════════════════════════════════════════════
    random_list = {
        20 = {  # char1 unlucky
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.10 }
            subtract_from_variable = { char1_power = temp_luck }
        }
        60 = {  # char1 average
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.30 }
            add_to_variable = { char1_power = temp_luck }
        }
        20 = {  # char1 critical
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.80 }
            add_to_variable = { char1_power = temp_luck }
        }
    }
    random_list = {
        20 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.10 }
            subtract_from_variable = { char2_power = temp_luck }
        }
        60 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.30 }
            add_to_variable = { char2_power = temp_luck }
        }
        20 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.80 }
            add_to_variable = { char2_power = temp_luck }
        }
    }

    # ═════════════════════════════════════════════════════════════
    # PHASE 3 — FORM CLASH (rock-paper-scissors)
    # ═════════════════════════════════════════════════════════════

    # Soresu dominates Ataru / Vaapad
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_soresu has_trait = form_soresu_mastery } }
            event_target:duel_char2_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char1_power = 6 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_soresu has_trait = form_soresu_mastery } }
            event_target:duel_char1_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char2_power = 6 }
    }

    # Djem So counter-strikes Ataru
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_djem_so has_trait = form_djem_so_mastery } }
            event_target:duel_char2_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char1_power = 4 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_djem_so has_trait = form_djem_so_mastery } }
            event_target:duel_char1_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char2_power = 4 }
    }

    # Makashi dominates plain duelists
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_makashi has_trait = form_makashi_mastery } }
            event_target:duel_char2_token = {
                NOT = { has_trait = form_makashi }
                NOT = { has_trait = form_makashi_mastery }
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_djem_so_mastery }
                NOT = { has_trait = form_juyo_vaapad }
            }
        }
        add_to_variable = { char1_power = 4 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_makashi has_trait = form_makashi_mastery } }
            event_target:duel_char1_token = {
                NOT = { has_trait = form_makashi }
                NOT = { has_trait = form_makashi_mastery }
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_djem_so_mastery }
                NOT = { has_trait = form_juyo_vaapad }
            }
        }
        add_to_variable = { char2_power = 4 }
    }

    # Shien deflects ranged attacks
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_shien has_trait = form_shien_mastery } }
            event_target:duel_char2_token = {
                OR = {
                    has_trait = blaster_duelist
                    has_trait = blaster_duelist_mastery

# a promotion replaces the previous rank (the rank traits have no on_add: the game validates it without a scope)
HOD_check_rank_promotion = {
	if = {
		limit = {
			has_trait = jedi_padawan_rank
			check_variable = { jedi_rank_xp > 499 }
		}
		remove_unit_leader_trait = jedi_padawan_rank
		add_unit_leader_trait = jedi_knight_rank
		set_variable = { jedi_rank_xp = 0 }
	}
	else_if = {
		limit = {
			has_trait = jedi_knight_rank
			check_variable = { jedi_rank_xp > 999 }
		}
		remove_unit_leader_trait = jedi_knight_rank
		add_unit_leader_trait = jedi_veteran_rank
		set_variable = { jedi_rank_xp = 0 }
	}
	else_if = {
		limit = {
			has_trait = dark_acolyte_rank
			check_variable = { jedi_rank_xp > 499 }
		}
		remove_unit_leader_trait = dark_acolyte_rank
		add_unit_leader_trait = dark_champion_rank
		set_variable = { jedi_rank_xp = 0 }
	}
}
