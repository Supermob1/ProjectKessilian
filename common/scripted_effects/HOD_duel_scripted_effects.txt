#####################################################################
# HOD duel — scripted effects
# ───────────────────────────────────────────────────────────────────
# 3 effects drive the duel flow:
#   HOD_duel_setup     — called in hod_duel.1 immediate; picks fighters,
#                        saves event targets, resets variables.
#   HOD_duel_advance   — called at the end of every hod_duel.2 option;
#                        either schedules the next round or resolution.
#   HOD_duel_cleanup   — clears every piece of duel state.
#
# Plus the unchanged XP/rank/mastery progression and outcome-style
# stamping effects, which are called from hod_duel.4.
#####################################################################

###############################
# HOD_duel_cleanup
# Wipes all duel state. Called from stand-down option, end of hod_duel.4,
# and (defensively) at the start of any new duel.
###############################
HOD_duel_cleanup = {
	clr_global_flag = hod_duel_active

}

###############################
# HOD_duel_setup
# Called in country scope (initiator), in hod_duel.1 immediate.
###############################
HOD_duel_setup = {
	hidden_effect = {
		set_global_flag = hod_duel_active
		random_army_leader = {
			save_event_target_as = duel_char1_token
			set_variable = { var = hod_duel_round value = 0 }
			# custom_effect_tooltip = allied_leader_tooltip
			HOD_duel_setup_calculate_powers = yes
		}
		random_enemy_country = {
			random_army_leader = {
				limit = {
					NOT = {
						OR = {
							AND = {
								event_target:duel_char1_token = { has_trait = hero_with_no_fear }
								has_trait = cybernetic_warlord
							}
							AND = {
								event_target:duel_char1_token = { has_trait = cybernetic_warlord }
								has_trait = hero_with_no_fear
							}
						}
					}
				}
				set_variable = { var = hod_duel_round value = 0 }
				save_event_target_as = duel_char2_token
				HOD_duel_setup_calculate_powers = yes
				# custom_effect_tooltip = enemy_leader_tooltip
			}
		}
		HOD_duel_init_round_state = yes
	}
	# custom_effect_tooltip = duel_tooltip
}

#####################################################################
# HOD_duel_calculation_base
# ───────────────────────────────────────────────────────────────────
# Per-round resolution. Computes char1_power and char2_power; the
# higher value is winning that round. Layered phases:
#   Phase 1 — Base contest: my att vs your def
#   Phase 2 — Luck roll: 20/60/20 bad/avg/crit, scales with luc
#   Phase 3 — Form clash: rock-paper-scissors lightsaber styles
#   Phase 4 — Force power interactions
#   Phase 5 — Non-Force & equipment interactions
#   Phase 6 — Rank pressure
# Call from event scope (initiator country). Reads variables from
# both fighter event targets.
#####################################################################
HOD_duel_calculation_base = {
    set_variable = { char1_power = 0 }
    set_variable = { char2_power = 0 }

    # ═════════════════════════════════════════════════════════════
    # PHASE 1 — BASE CONTEST
    # My attack potential, soaked by half of opponent's defense.
    # ═════════════════════════════════════════════════════════════
    add_to_variable = { char1_power = event_target:duel_char1_token.att }
    set_temp_variable = { temp_soak = event_target:duel_char2_token.def }
    multiply_temp_variable = { temp_soak = 0.5 }
    subtract_from_variable = { char1_power = temp_soak }

    add_to_variable = { char2_power = event_target:duel_char2_token.att }
    set_temp_variable = { temp_soak = event_target:duel_char1_token.def }
    multiply_temp_variable = { temp_soak = 0.5 }
    subtract_from_variable = { char2_power = temp_soak }

    # ═════════════════════════════════════════════════════════════
    # PHASE 2 — LUCK ROLL (per character)
    # 20% unlucky / 60% average / 20% critical hit.
    # Higher luc = bigger swing in both directions.
    # ═════════════════════════════════════════════════════════════
    random_list = {
        20 = {  # char1 unlucky
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.10 }
            subtract_from_variable = { char1_power = temp_luck }
        }
        60 = {  # char1 average
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.30 }
            add_to_variable = { char1_power = temp_luck }
        }
        20 = {  # char1 critical
            set_temp_variable = { temp_luck = event_target:duel_char1_token.luc }
            multiply_temp_variable = { temp_luck = 0.80 }
            add_to_variable = { char1_power = temp_luck }
        }
    }
    random_list = {
        20 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.10 }
            subtract_from_variable = { char2_power = temp_luck }
        }
        60 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.30 }
            add_to_variable = { char2_power = temp_luck }
        }
        20 = {
            set_temp_variable = { temp_luck = event_target:duel_char2_token.luc }
            multiply_temp_variable = { temp_luck = 0.80 }
            add_to_variable = { char2_power = temp_luck }
        }
    }

    # ═════════════════════════════════════════════════════════════
    # PHASE 3 — FORM CLASH (rock-paper-scissors)
    # ═════════════════════════════════════════════════════════════

    # Soresu dominates Ataru / Vaapad
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_soresu has_trait = form_soresu_mastery } }
            event_target:duel_char2_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char1_power = 6 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_soresu has_trait = form_soresu_mastery } }
            event_target:duel_char1_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char2_power = 6 }
    }

    # Djem So counter-strikes Ataru
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_djem_so has_trait = form_djem_so_mastery } }
            event_target:duel_char2_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char1_power = 4 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_djem_so has_trait = form_djem_so_mastery } }
            event_target:duel_char1_token = { OR = { has_trait = form_ataru has_trait = form_ataru_mastery has_trait = form_juyo_vaapad } }
        }
        add_to_variable = { char2_power = 4 }
    }

    # Makashi dominates plain duelists
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_makashi has_trait = form_makashi_mastery } }
            event_target:duel_char2_token = {
                NOT = { has_trait = form_makashi }
                NOT = { has_trait = form_makashi_mastery }
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_djem_so_mastery }
                NOT = { has_trait = form_juyo_vaapad }
            }
        }
        add_to_variable = { char1_power = 4 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_makashi has_trait = form_makashi_mastery } }
            event_target:duel_char1_token = {
                NOT = { has_trait = form_makashi }
                NOT = { has_trait = form_makashi_mastery }
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_djem_so_mastery }
                NOT = { has_trait = form_juyo_vaapad }
            }
        }
        add_to_variable = { char2_power = 4 }
    }

    # Shien deflects ranged attacks
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_shien has_trait = form_shien_mastery } }
            event_target:duel_char2_token = {
                OR = {
                    has_trait = blaster_duelist
                    has_trait = blaster_duelist_mastery
                    has_trait = rapid_fire_barrage
                    has_trait = targeting_computer
                    has_trait = thermal_detonators
                }
            }
        }
        add_to_variable = { char1_power = 5 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_shien has_trait = form_shien_mastery } }
            event_target:duel_char1_token = {
                OR = {
                    has_trait = blaster_duelist
                    has_trait = blaster_duelist_mastery
                    has_trait = rapid_fire_barrage
                    has_trait = targeting_computer
                    has_trait = thermal_detonators
                }
            }
        }
        add_to_variable = { char2_power = 5 }
    }

    # Jar'Kai overwhelms single-blade defenders
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = form_jar_kai has_trait = form_jar_kai_mastery } }
            event_target:duel_char2_token = {
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_jar_kai }
                NOT = { has_trait = form_jar_kai_mastery }
            }
        }
        add_to_variable = { char1_power = 3 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { OR = { has_trait = form_jar_kai has_trait = form_jar_kai_mastery } }
            event_target:duel_char1_token = {
                NOT = { has_trait = form_soresu_mastery }
                NOT = { has_trait = form_jar_kai }
                NOT = { has_trait = form_jar_kai_mastery }
            }
        }
        add_to_variable = { char2_power = 3 }
    }

    # Niman + battle_meditation synergy (the diplomat-soldier)
    if = {
        limit = {
            event_target:duel_char1_token = {
                OR = { has_trait = form_niman has_trait = form_niman_mastery }
                has_trait = battle_meditation
            }
        }
        add_to_variable = { char1_power = 3 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = {
                OR = { has_trait = form_niman has_trait = form_niman_mastery }
                has_trait = battle_meditation
            }
        }
        add_to_variable = { char2_power = 3 }
    }

    # Shii-Cho mastery is "fundamentals refined" — small all-round bonus
    if = {
        limit = { event_target:duel_char1_token = { has_trait = form_shii_cho_mastery } }
        add_to_variable = { char1_power = 2 }
    }
    if = {
        limit = { event_target:duel_char2_token = { has_trait = form_shii_cho_mastery } }
        add_to_variable = { char2_power = 2 }
    }

    # ═════════════════════════════════════════════════════════════
    # PHASE 4 — FORCE POWER INTERACTIONS
    # ═════════════════════════════════════════════════════════════

    # Force Barrier hard-counters Lightning
    if = {
        limit = {
            event_target:duel_char1_token = { has_trait = force_barrier }
            event_target:duel_char2_token = { OR = { has_trait = force_lightning has_trait = force_lightning_mastery } }
        }
        add_to_variable = { char1_power = 5 }
        subtract_from_variable = { char2_power = 3 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { has_trait = force_barrier }
            event_target:duel_char1_token = { OR = { has_trait = force_lightning has_trait = force_lightning_mastery } }
        }
        add_to_variable = { char2_power = 5 }
        subtract_from_variable = { char1_power = 3 }
    }

    # Lightning vs non-barrier — big damage
    if = {
        limit = {
            event_target:duel_char1_token = { has_trait = force_lightning_mastery }
            event_target:duel_char2_token = { NOT = { has_trait = force_barrier } }
        }
        add_to_variable = { char1_power = 4 }
    }
    if = {
        limit = {
            event_target:duel_char2_token = { has_trait = force_lightning_mastery }
            event_target:duel_char1_token = { NOT = { has_trait = force_barrier } }
        }
        add_to_variable = { char2_power = 4 }
    }

    # Prescience reads the unsighted (asymmetric)
    if = {
        limit = {
            event_target:duel_char1_token = { OR = { has_trait = force_prescience has_trait = force_prescience_mastery } }

                NOT = { has_trait = force_prescience }

            event_target:duel_char2_token = { NOT = { has_trait = force_speed } }

        limit = {
            event_target:duel_char2_token = { has_trait = force_speed }

        limit = { event_target:duel_char1_token = { has_trait = force_push } }

#####################################################################
# Round-resolution flow (CK3-style turn system)
# ───────────────────────────────────────────────────────────────────
# Suggested event chain:
#   hod_duel.1   challenge → fires hod_duel.2 + hod_duel.101
#   hod_duel.2   char1's turn (player). Each option calls
#                HOD_duel_set_move_* on duel_char1_token, then
#                after-block fires hod_duel.201 for char2's owner.
#   hod_duel.201 char2's turn (AI). Mirrors hod_duel.2 but on
#                duel_char2_token; after fires hod_duel.3.
#   hod_duel.3   resolver (hidden). immediate calls
#                HOD_duel_resolve_round. after calls
#                HOD_duel_route_next.
#
# Variables this layer manages:
#   On each character (set via event_target scope):
#     duel_move        1 = attack, 2 = defend, 3 = counter
#     rounds_won       running tally
#     wounds           damage taken
#     last_flavor_id   most recent flavor ID (loc reads this)
#     hod_duel_round   round counter (incremented at resolution)
#   On the initiator country:
#     char1_power, char2_power  computed each round
#     last_round_winner         0 = tie, 1 = char1, 2 = char2
#     duel_outcome              0 = ongoing, 1 = char1, 2 = char2, 3 = stalemate
#####################################################################


###############################
# HOD_duel_init_round_state
# Initialize per-duel tracking. Call once at duel start
# (e.g. at the end of HOD_duel_setup), initiator country scope.
###############################
HOD_duel_init_round_state = {
    event_target:duel_char1_token = {
        set_variable = { duel_move = 0 }
        set_variable = { rounds_won = 0 }
        set_variable = { wounds = 0 }
        set_variable = { last_flavor_id = 0 }
    }
    event_target:duel_char2_token = {
        set_variable = { duel_move = 0 }
        set_variable = { rounds_won = 0 }
        set_variable = { wounds = 0 }
        set_variable = { last_flavor_id = 0 }
    }
    set_variable = { last_round_winner = 0 }
    set_variable = { round_decisive = 0 }
    set_variable = { duel_outcome = 0 }

    # Per-round narrative snapshots (3 rounds total)
    set_variable = { round1_winner = 0 }
    set_variable = { round1_decisive = 0 }
    set_variable = { round1_char1_move = 0 }
    set_variable = { round1_char2_move = 0 }
    set_variable = { round1_char1_flavor = 0 }
    set_variable = { round1_char2_flavor = 0 }
    set_variable = { round2_winner = 0 }
    set_variable = { round2_decisive = 0 }
    set_variable = { round2_char1_move = 0 }
    set_variable = { round2_char2_move = 0 }
    set_variable = { round2_char1_flavor = 0 }
    set_variable = { round2_char2_flavor = 0 }
    set_variable = { round3_winner = 0 }
    set_variable = { round3_decisive = 0 }
    set_variable = { round3_char1_move = 0 }
    set_variable = { round3_char2_move = 0 }
    set_variable = { round3_char1_flavor = 0 }
    set_variable = { round3_char2_flavor = 0 }
}

###############################
# HOD_duel_resolve_round
# Call from hod_duel.3 immediate, initiator country scope.
# Both moves already recorded; this:
#   1. Runs HOD_duel_calculation_base (att/def/luck/trait phases)
#   2. Applies CK3-style move interaction (rock-paper-scissors, +8)
#   3. Determines round winner; loser takes a wound, winner +1 round
#   4. Decisive margin (>15) deals an extra wound
#   5. Advances round counter on both characters
#   6. Calls HOD_duel_check_end to set duel_outcome
#
# Move interaction:
#   Attack(1)  > Defend(2)   — pushes through the guard
#   Defend(2)  > Counter(3)  — weathers the counter
#   Counter(3) > Attack(1)   — punishes the committed strike
###############################
HOD_duel_resolve_round = {
	hidden_effect = {
		HOD_duel_calculation_base = yes
	}
	custom_effect_tooltip = "HELLO ?"
    # ── Move interaction (rock-paper-scissors) ──────────────────
    # char1 wins matchup
    if = {
        limit = {
            check_variable = { event_target:duel_char1_token.duel_move = 1 }
            check_variable = { event_target:duel_char2_token.duel_move = 2 }
        }
        add_to_variable = { char1_power = 8 }
		custom_effect_tooltip = "CHAR 1 ATTACK CHAR 2 DEFEND"
    }
    if = {
        limit = {
            check_variable = { event_target:duel_char1_token.duel_move = 2 }
            check_variable = { event_target:duel_char2_token.duel_move = 3 }
        }
        add_to_variable = { char1_power = 8 }
		custom_effect_tooltip = "CHAR 1 DEFEND CHAR 2 COUNTER"
    }
    if = {
        limit = {
            check_variable = { event_target:duel_char1_token.duel_move = 3 }
            check_variable = { event_target:duel_char2_token.duel_move = 1 }
        }
        add_to_variable = { char1_power = 8 }
		custom_effect_tooltip = "CHAR 1 COUNTER CHAR 2 ATTACK"
    }
    # char2 wins matchup
    if = {
        limit = {
            check_variable = { event_target:duel_char2_token.duel_move = 1 }
            check_variable = { event_target:duel_char1_token.duel_move = 2 }
        }
        add_to_variable = { char2_power = 8 }
		custom_effect_tooltip = "CHAR 2 ATTACK CHAR 1 DEFEND"
    }
    if = {
        limit = {
            check_variable = { event_target:duel_char2_token.duel_move = 2 }
            check_variable = { event_target:duel_char1_token.duel_move = 3 }
        }
        add_to_variable = { char2_power = 8 }
		custom_effect_tooltip = "CHAR 2 DEFEND CHAR 1 COUNTER"
    }
    if = {
        limit = {
            check_variable = { event_target:duel_char2_token.duel_move = 3 }
            check_variable = { event_target:duel_char1_token.duel_move = 1 }
        }
        add_to_variable = { char2_power = 8 }
		custom_effect_tooltip = "CHAR 2 COUNTER CHAR 1 ATTACK"
    }

    # ── Determine round winner ──────────────────────────────────
    set_temp_variable = { temp_diff = char1_power }
    subtract_from_temp_variable = { temp_diff = char2_power }
    set_variable = { round_decisive = 0 }

    if = {
        limit = { check_variable = { temp_diff > 0 } }
        set_variable = { last_round_winner = 1 }
        event_target:duel_char1_token = { add_to_variable = { rounds_won = 1 } }
        event_target:duel_char2_token = { add_to_variable = { wounds = 1 } }
        if = {
            limit = { check_variable = { temp_diff > 15 } }
            event_target:duel_char2_token = { add_to_variable = { wounds = 1 } }
            set_variable = { round_decisive = 1 }
        }
    }
    else_if = {
        limit = { check_variable = { temp_diff < 0 } }
        set_variable = { last_round_winner = 2 }
        event_target:duel_char2_token = { add_to_variable = { rounds_won = 1 } }
        event_target:duel_char1_token = { add_to_variable = { wounds = 1 } }
        if = {
            limit = { check_variable = { temp_diff < -15 } }
            event_target:duel_char1_token = { add_to_variable = { wounds = 1 } }
            set_variable = { round_decisive = 1 }
        }
    }
    else = {
        set_variable = { last_round_winner = 0 }
    }

    # Advance round counter on both
    event_target:duel_char1_token = { add_to_variable = { hod_duel_round = 1 } }
    event_target:duel_char2_token = { add_to_variable = { hod_duel_round = 1 } }

    # Snapshot this round's data into round1_/round2_/round3_ vars
    # so paragraphs can describe the duel's history after the fact.
    HOD_duel_snapshot_round = yes

    HOD_duel_check_end = yes
}


###############################
# HOD_duel_snapshot_round
# Saves last_round_winner / round_decisive / both fighter moves &
# flavor IDs into the appropriate roundN_* slot. Reads hod_duel_round
# (already incremented), so:
#   hod_duel_round = 1 → just finished round 1
#   hod_duel_round = 2 → just finished round 2
#   hod_duel_round = 3 → just finished round 3 (finale)
###############################
HOD_duel_snapshot_round = {
    if = {
        limit = { check_variable = { event_target:duel_char1_token.hod_duel_round = 1 } }
        set_variable = { round1_winner = last_round_winner }
        set_variable = { round1_decisive = round_decisive }
        set_variable = { round1_char1_move = event_target:duel_char1_token.duel_move }
        set_variable = { round1_char2_move = event_target:duel_char2_token.duel_move }
        set_variable = { round1_char1_flavor = event_target:duel_char1_token.last_flavor_id }
        set_variable = { round1_char2_flavor = event_target:duel_char2_token.last_flavor_id }
    }
    else_if = {
        limit = { check_variable = { event_target:duel_char1_token.hod_duel_round = 2 } }
        set_variable = { round2_winner = last_round_winner }
        set_variable = { round2_decisive = round_decisive }
        set_variable = { round2_char1_move = event_target:duel_char1_token.duel_move }
        set_variable = { round2_char2_move = event_target:duel_char2_token.duel_move }
        set_variable = { round2_char1_flavor = event_target:duel_char1_token.last_flavor_id }
        set_variable = { round2_char2_flavor = event_target:duel_char2_token.last_flavor_id }
    }
    else_if = {
        limit = { check_variable = { event_target:duel_char1_token.hod_duel_round = 3 } }
        set_variable = { round3_winner = last_round_winner }
        set_variable = { round3_decisive = round_decisive }
        set_variable = { round3_char1_move = event_target:duel_char1_token.duel_move }
        set_variable = { round3_char2_move = event_target:duel_char2_token.duel_move }
        set_variable = { round3_char1_flavor = event_target:duel_char1_token.last_flavor_id }
        set_variable = { round3_char2_flavor = event_target:duel_char2_token.last_flavor_id }
    }
}


###############################
# HOD_duel_check_end
# Fixed 3 rounds. After round 3 resolves, decide winner by rounds_won.
#   0 = ongoing
#   1 = char1 wins (more rounds)
#   2 = char2 wins (more rounds)
#   3 = stalemate (equal rounds)
###############################
HOD_duel_check_end = {
    if = {
        limit = { check_variable = { event_target:duel_char1_token.hod_duel_round > 2 } }
        if = {
            limit = {
                check_variable = { event_target:duel_char1_token.rounds_won > event_target:duel_char2_token.rounds_won }
            }
            set_variable = { duel_outcome = 1 }
        }
        else_if = {
            limit = {
                check_variable = { event_target:duel_char2_token.rounds_won > event_target:duel_char1_token.rounds_won }
            }
            set_variable = { duel_outcome = 2 }
        }
        else = {
            set_variable = { duel_outcome = 3 }
        }
    }
    else = {
        set_variable = { duel_outcome = 0 }
    }
}


###############################
# HOD_duel_route_next
# Call from hod_duel.3 after-block, initiator country scope.
# Branches based on duel_outcome:
#   0 → next round (re-fire hod_duel.2 to char1's owner)
#   1 → victory event for char1, defeat for char2; cleanup
#   2 → defeat for char1, victory for char2; cleanup
#   3 → stalemate for both; cleanup
###############################
HOD_duel_route_next = {
    if = {
        limit = { check_variable = { duel_outcome = 0 } }
        # Mid-duel — next round, char1 picks again
        event_target:duel_char1_token.owner = { country_event = { id = hod_duel.2 days = 1 } }
		
		custom_effect_tooltip = "NEXT event"
    }
    else = {
        # Duel finished — summary view ONLY for char1's owner.
        # Do NOT also fire hod_duel.201 — it would race with hod_duel.2.
        # When char1 clicks Acknowledge, finalize_outcome runs cleanup
        # which zeros duel_outcome. Then char2's hod_duel.201 would see
        # duel_outcome=0, re-enable options A/B/C, AI picks one, fires
        # hod_duel.3, and the duel loops forever.
        # char2's owner gets the result through hod_duel.20/.21/.22
        # unit_leader_events that finalize_outcome fires.
        event_target:duel_char1_token.owner = { country_event = { id = hod_duel.2 days = 1 } }
		custom_effect_tooltip = "DUEL FINISHED"
    }
}


###############################
# HOD_duel_finalize_outcome
# Called by the Acknowledge option in the summary view.
# Fires per-fighter outcome events, stamps style flags, awards XP,
# then cleans up. Runs in initiator country scope.
###############################
HOD_duel_finalize_outcome = {

HOD_duel_advance = {
	if = {
		limit = {
			OR = {
				check_variable = { hod_duel_edge > 4 }
				check_variable = { hod_duel_edge < -4 }
				check_variable = { hod_duel_round > 3 }
			}

		country_event = { id = hod_duel.4 days = 1 }
	}
	else = {

HOD_check_rank_promotion = {
	if = {
		limit = {
			has_trait = jedi_padawan_rank
			check_variable = { jedi_rank_xp > 499 }
		}
		add_unit_leader_trait = jedi_knight_rank
		set_variable = { jedi_rank_xp = 0 }
	}
	else_if = {
		limit = {
			has_trait = jedi_knight_rank
			check_variable = { jedi_rank_xp > 999 }
		}
		add_unit_leader_trait = jedi_veteran_rank
		set_variable = { jedi_rank_xp = 0 }
	}
	else_if = {
		limit = {
			has_trait = dark_acolyte_rank
			check_variable = { jedi_rank_xp > 499 }
		}
		add_unit_leader_trait = dark_champion_rank
		set_variable = { jedi_rank_xp = 0 }
	}
}
