Clone_wars_start = {
	CIS = { 
		# the war is declared below, by game mode
		set_cosmetic_tag = CIS_Confderation_of_Independent_Systems
		# load_oob = "HOD_Droid_Army"

		# Geonosis is normally the Confederacy's from the start (game_mode_history.py); this catches a released GEO
		if = {
			limit = { country_exists = GEO NOT = { has_global_flag = HOD_battle_royale } }
			GEO = {
				every_owned_state = {
					add_core_of = CIS
				}
			}
			annex_country = {
				target = GEO
				transfer_troops = yes
			}
		}
		REP = { 
			# load_oob = "HOD_Grande_Armee"
			set_politics = {
				ruling_party = unitist
				elections_allowed = no
			}
			hidden_effect = {
				add_timed_idea = {
					idea = Unitism_drift
					days = 180
				}
				# REP_unexpected_attack (-50% army for 90 days) removed: the Republic is the attacker at Geonosis
			}
			add_country_leader_role = {
				character = REP_sheev_palpatine
				promote_leader = yes
				country_leader = {
					ideology = unitist_subtype
					traits = { shrewd_chancellor }
				}
			}
		}
		# add_ideas = CIS_Great_Droid_Army
		# load_oob = "HOD_CIS_naval"
	}
	# REP = { load_oob = "HOD_REP_naval" }

	# Total War: one open war to capitulation; Battle Royale: no Republic-Confederacy war, every planet fights alone
	if = {
		limit = { NOT = { has_global_flag = HOD_battle_royale } }
		CIS = { declare_war_on = { target = REP type = annex_everything } }
	}
}
