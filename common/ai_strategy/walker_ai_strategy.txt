# Use the total-conversion/native armor role so existing division demand also creates walkers.
# Shift armored production to walkers only after the first project and sufficient industry.

    enable = {
        is_ai = yes
        has_tech = rep_walker_atpt
        num_of_military_factories > 24
        OR = {
            has_tech = rep_walker_atpt
            has_tech = rep_walker_atpt
            has_tech = rep_walker_atte
            has_tech = rep_walker_atte
            has_tech = rep_walker_atap
            has_tech = rep_walker_atap
        }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = role_ratio id = armor value = 12 }
    ai_strategy = { type = land_xp_spend_priority id = unlock_doctrine value = 40 }
    ai_strategy = { type = land_xp_spend_priority id = division_template value = 100 }
    ai_strategy = { type = land_xp_spend_priority id = equipment_variant value = 100 }
    ai_strategy = { type = equipment_production_min_factories_archetype id = walker_chassis value = 1 }
}
