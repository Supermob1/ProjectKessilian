generic_debug = {

	# Debug: start a duel now between a random duelist of ours and an enemy one (HOD_duel_fight), anywhere on the map.
	HOD_debug_duel_random = {
		cost = 0
		fire_only_once = no
		visible = {
			has_war = yes
		}
		available = {
			any_army_leader = { HOD_is_duelist = yes }
			any_enemy_country = { any_army_leader = { HOD_is_duelist = yes } }
		}
		icon = generic_spy
		complete_effect = {
			# debug: a duel between one of our duelists and an enemy one, wherever they are
			random_army_leader = {
				limit = { HOD_is_duelist = yes }
				save_event_target_as = duel_char1_token
			}
			random_enemy_country = {
				random_army_leader = {
					limit = { HOD_is_duelist = yes }
					save_event_target_as = duel_char2_token
				}
			}
			if = {
				limit = { has_event_target = duel_char1_token has_event_target = duel_char2_token }
				HOD_duel_fight = yes
			}
		}
	}

	# ---- HOD test kit for the operation system: one closable hyperlane + a garrison fleet ----
	# Debug: CIS and the Republic go to war (if they are not already), to test the lane and the fleet
	HOD_debug_test_war = {
		cost = 0
		fire_only_once = no
		icon = generic_operation
		available = { CIS = { NOT = { has_war_with = REP } } }
		complete_effect = {
			CIS = { declare_war_on = { target = REP type = annex_everything } }
		}
	}
	# Debug: close the Coruscant - Borleias hyperlane to enemy and neutral armies (own side can still use it)
	HOD_debug_close_test_lane = {
		cost = 0
		fire_only_once = no
		icon = generic_operation
		available = { NOT = { has_global_flag = HOD_test_lane_closed } }
		complete_effect = { set_global_flag = HOD_test_lane_closed }
	}
	# Debug: open it again
	HOD_debug_open_test_lane = {
		cost = 0
		fire_only_once = no
		icon = generic_operation
		available = { has_global_flag = HOD_test_lane_closed }
		complete_effect = { clr_global_flag = HOD_test_lane_closed }
	}
	# Debug: park a Republic garrison fleet in space next to Coruscant (Coruscant Sector), to test if it blocks invasions
	HOD_debug_spawn_garrison_fleet = {
		cost = 0
		fire_only_once = no
		icon = generic_operation
		complete_effect = { REP = { load_oob = "HOD_test_space_garrison" } }
	}
	# Debug: white peace between CIS and the Republic
	HOD_debug_test_peace = {
		cost = 0
		fire_only_once = no
		icon = generic_operation
		available = { CIS = { has_war_with = REP } }
		complete_effect = { CIS = { white_peace = REP } }
	}
}
