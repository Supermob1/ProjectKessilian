###############################################################
# HOD Duel — Scripted Localization
# ─────────────────────────────────────────────────────────────
# The 5-paragraph narrative arc:
#   Para 1 — atmosphere (the meeting)
#   Para 2 — first exchange (round 1 — fills after round 1 resolves)
#   Para 3 — second exchange (round 2 — fills after round 2 resolves)
#   Para 4 — finale (round 3 — fills after round 3 resolves; names winner)
#   Para 5 — consequences (wounds, aftermath; fills after round 3)
#
# Display timing (hod_duel.2 / hod_duel.201 fire each round + once
# for summary):
#   Round 1 view (hod_duel_round=0): Para 1 only
#   Round 2 view (hod_duel_round=1): Para 1 + 2
#   Round 3 view (hod_duel_round=2): Para 1 + 2 + 3
#   Summary view (hod_duel_round=3): Para 1 + 2 + 3 + 4 + 5
###############################################################


###############################################################
# DuelParagraph1 — atmosphere
###############################################################
defined_text = {
	name = DuelParagraph1
	text = {
		trigger = {
			event_target:duel_char1_token = { has_trait = force_sensitive }
			event_target:duel_char2_token = { has_trait = force_sensitive }
		}
		localization_key = hod_duel.2.d_force_vs_force
	}
	text = {
		trigger = {
			event_target:duel_char1_token = { has_trait = force_sensitive }
			event_target:duel_char2_token = { NOT = { has_trait = force_sensitive } }
		}
		localization_key = hod_duel.2.d_force_vs_warrior
	}
	text = {
		trigger = {
			event_target:duel_char1_token = { NOT = { has_trait = force_sensitive } }
			event_target:duel_char2_token = { has_trait = force_sensitive }
		}
		localization_key = hod_duel.2.d_warrior_vs_force
	}
	text = {
		trigger = {
			event_target:duel_char1_token = { NOT = { has_trait = force_sensitive } }
			event_target:duel_char2_token = { NOT = { has_trait = force_sensitive } }
		}
		localization_key = hod_duel.2.d_warrior_vs_warrior
	}
}

# DuelParagraph2 — first exchange (round 1 recap)
# Empty until round 1 has resolved (hod_duel_round >= 1).
###############################################################
defined_text = {
	name = DuelParagraph2

	# Round 1 hasn't resolved yet
	text = {
		trigger = { check_variable = { event_target:duel_char1_token.hod_duel_round < 1 } }
		localization_key = hod_duel_p_empty
	}

	# Tied opening
	text = {
		trigger = { check_variable = { round1_winner = 0 } }
		localization_key = hod_duel_p2_tied
	}
	# char1 wins round 1, decisive
	text = {
		trigger = {
			check_variable = { round1_winner = 1 }
			check_variable = { round1_decisive = 1 }
		}
		localization_key = hod_duel_p2_char1_decisive
	}
	# char1 wins round 1 by move type
	text = {
		trigger = {
			check_variable = { round1_winner = 1 }
			check_variable = { round1_char1_move = 1 }
		}
		localization_key = hod_duel_p2_char1_attack
	}
	text = {
		trigger = {
			check_variable = { round1_winner = 1 }
			check_variable = { round1_char1_move = 2 }
		}
		localization_key = hod_duel_p2_char1_defend
	}
	text = {
		trigger = {
			check_variable = { round1_winner = 1 }
			check_variable = { round1_char1_move = 3 }
		}
		localization_key = hod_duel_p2_char1_counter
	}
	# char2 wins round 1, decisive
	text = {
		trigger = {
			check_variable = { round1_winner = 2 }
			check_variable = { round1_decisive = 1 }
		}
		localization_key = hod_duel_p2_char2_decisive
	}
	text = {
		trigger = {
			check_variable = { round1_winner = 2 }
			check_variable = { round1_char2_move = 1 }
		}
		localization_key = hod_duel_p2_char2_attack
	}
	text = {
		trigger = {
			check_variable = { round1_winner = 2 }
			check_variable = { round1_char2_move = 2 }
		}
		localization_key = hod_duel_p2_char2_defend
	}
	text = {
		trigger = {
			check_variable = { round1_winner = 2 }
			check_variable = { round1_char2_move = 3 }
		}
		localization_key = hod_duel_p2_char2_counter
	}
}

###############################################################
# DuelParagraph4 — finale (round 3 outcome)
# Empty until round 3 has resolved (hod_duel_round >= 3).
# This paragraph reveals the duel winner.
###############################################################
defined_text = {
	name = DuelParagraph4

	text = {
		trigger = { check_variable = { event_target:duel_char1_token.hod_duel_round < 3 } }
		localization_key = hod_duel_p_empty
	}

	# Overall duel winner = duel_outcome (1/2/3)
	# Round 3 outcome adds drama (round3_winner)
	text = {
		trigger = {
			check_variable = { duel_outcome = 1 }
			check_variable = { round3_decisive = 1 }
		}
		localization_key = hod_duel_p4_char1_finale_decisive
	}
	text = {
		trigger = { check_variable = { duel_outcome = 1 } }
		localization_key = hod_duel_p4_char1_finale
	}
	text = {
		trigger = {
			check_variable = { duel_outcome = 2 }
			check_variable = { round3_decisive = 1 }
		}
		localization_key = hod_duel_p4_char2_finale_decisive
	}
	text = {
		trigger = { check_variable = { duel_outcome = 2 } }
		localization_key = hod_duel_p4_char2_finale
	}
	text = {
		trigger = { check_variable = { duel_outcome = 3 } }
		localization_key = hod_duel_p4_stalemate
	}
}


###############################################################
# DuelParagraph5 — consequences (post-duel state)
# Empty until round 3 has resolved.
# Describes wound state of both fighters.
###############################################################
defined_text = {
	name = DuelParagraph5

	text = {
		trigger = { check_variable = { event_target:duel_char1_token.hod_duel_round < 3 } }
		localization_key = hod_duel_p_empty
	}

	# Stalemate, both wounded
	text = {
		trigger = {
			check_variable = { duel_outcome = 3 }
			check_variable = { event_target:duel_char1_token.wounds > 0 }
			check_variable = { event_target:duel_char2_token.wounds > 0 }
		}
		localization_key = hod_duel_p5_stale_both_hurt
	}
	# Stalemate, clean
	text = {
		trigger = { check_variable = { duel_outcome = 3 } }
		localization_key = hod_duel_p5_stale_clean
	}
	# char1 wins, char1 also wounded — pyrrhic
	text = {
		trigger = {
			check_variable = { duel_outcome = 1 }
			check_variable = { event_target:duel_char1_token.wounds > 1 }
		}
		localization_key = hod_duel_p5_char1_pyrrhic
	}
	# char1 wins, char2 severely wounded
	text = {
		trigger = {
			check_variable = { duel_outcome = 1 }
			check_variable = { event_target:duel_char2_token.wounds > 2 }
		}
		localization_key = hod_duel_p5_char1_wins_severe
	}
	# char1 wins, char2 wounded
	text = {
		trigger = {
			check_variable = { duel_outcome = 1 }
			check_variable = { event_target:duel_char2_token.wounds > 0 }
		}
		localization_key = hod_duel_p5_char1_wins_wounded
	}
	# char1 wins, clean
	text = {
		trigger = { check_variable = { duel_outcome = 1 } }
		localization_key = hod_duel_p5_char1_wins_clean
	}
	# char2 wins, char2 also wounded — pyrrhic
	text = {
		trigger = {
			check_variable = { duel_outcome = 2 }
			check_variable = { event_target:duel_char2_token.wounds > 1 }
		}
		localization_key = hod_duel_p5_char2_pyrrhic
	}
	# char2 wins, char1 severely wounded
	text = {
		trigger = {
			check_variable = { duel_outcome = 2 }
			check_variable = { event_target:duel_char1_token.wounds > 2 }
		}
		localization_key = hod_duel_p5_char2_wins_severe
	}
	# char2 wins, char1 wounded
	text = {
		trigger = {
			check_variable = { duel_outcome = 2 }
			check_variable = { event_target:duel_char1_token.wounds > 0 }
		}
		localization_key = hod_duel_p5_char2_wins_wounded
	}
	# char2 wins, clean
	text = {
		trigger = { check_variable = { duel_outcome = 2 } }
		localization_key = hod_duel_p5_char2_wins_clean
	}
}
